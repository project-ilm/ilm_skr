/* ترتیب شدہ صف میں ثنائی تلاش */
صحيح ثنائی_تلاش(صحيح اقدار[], صحيح گنتی, صحيح کلید) {
  صحيح کم = 0, بلند = گنتی - 1;
  طالما (کم <= بلند) {
    صحيح درمیان = کم + (بلند - کم) / 2;
    إذا (اقدار[درمیان] == کلید) أرجع درمیان;
    إذا (اقدار[درمیان] < کلید) کم = درمیان + 1; وإلا بلند = درمیان - 1;
  }
  أرجع -1;
}
صحيح مکھ(فراغ) {
  صحيح اقدار[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
  printf("%d %d\n", ثنائی_تلاش(اقدار, 10, 23), ثنائی_تلاش(اقدار, 10, 7));
  أرجع 0;
}

  بنية درخت_گرہ { صحيح کلید; درخت_گرہ *بایاں = عدم, *دایاں = عدم; };
  درخت_گرہ *جڑ = عدم;
  درخت_گرہ *داخل_کرو(درخت_گرہ *گرہ, صحيح کلید) { إذا (!گرہ) { گرہ = جديد درخت_گرہ; گرہ->کلید = کلید; أرجع گرہ; } إذا (کلید < گرہ->کلید) گرہ->بایاں = داخل_کرو(گرہ->بایاں, کلید); وإلا گرہ->دایاں = داخل_کرو(گرہ->دایاں, کلید); أرجع گرہ; }
  فراغ درمیانی_ترتیب(درخت_گرہ *گرہ) { إذا (!گرہ) أرجع; درمیانی_ترتیب(گرہ->بایاں); std::cout << گرہ->کلید << " "; درمیانی_ترتیب(گرہ->دایاں); }

  فراغ داخل_کرو(صحيح کلید) { جڑ = داخل_کرو(جڑ, کلید); }
  فراغ print() { درمیانی_ترتیب(جڑ); std::cout << std::endl; }
};

// ڈھیر کی جماعت، جاوا میں 
عام صنف ڈھیر {
  خاص صحيح[] اشیاء = جديد صحيح[16];
  خاص صحيح چوٹی = 0;
  عام فراغ ڈالو(صحيح قدر) { اشیاء[چوٹی++] = قدر; }
  عام صحيح نکالو() { أرجع اشیاء[--چوٹی]; }
  عام ساكن فراغ مکھ(نص[] args) {
    ڈھیر اشیاء = جديد ڈھیر();
    لكل (صحيح قدر = 1; قدر <= 3; قدر++) اشیاء.ڈالو(قدر * 10);
    System.out.println(اشیاء.نکالو() + " " + اشیاء.نکالو());
  }
}
